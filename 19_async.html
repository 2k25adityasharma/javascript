// ============================== // Async JavaScript Quick Notes // ============================== // setTimeout() -> Runs once after a delay setTimeout(() => { console.log("Hello"); }, 2000); // clearTimeout() -> Cancels a scheduled timeout const timer = setTimeout(() => { console.log("Won't Execute"); }, 3000); clearTimeout(timer); // setInterval() -> Runs repeatedly after every delay const interval = setInterval(() => { console.log("Running..."); }, 1000); // clearInterval() -> Stops a running interval clearInterval(interval); // Callback Function -> Function passed as an argument function greet(name, callback) { console.log("Hello " + name); callback(); } function bye() { console.log("Bye"); } greet("Aditya", bye); // Promise -> Handles asynchronous operations const promise = new Promise((resolve, reject) => { let success = true; if (success) resolve("Success"); else reject("Failed"); }); promise .then((data) => console.log(data)) .catch((err) => console.log(err)); // fetch() -> Used to request data from an API fetch("https://jsonplaceholder.typicode.com/users") .then((res) => res.json()) .then((data) => console.log(data)) .catch((err) => console.log(err)); // async -> Function automatically returns a Promise async function hello() { return "Hello"; } // await -> Waits for a Promise to complete async function getUsers() { try { const response = await fetch("https://jsonplaceholder.typicode.com/users"); const data = await response.json(); console.log(data); } catch (error) { console.log(error); } } getUsers();
